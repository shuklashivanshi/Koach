POST /v1/optionals/stress body kind 'hours' | 'regret' with slug, or kind 'answer' with slug, unit and optional text (10 to 600 chars) returns { kind, title, body, verdict: 'comfortable' | 'tight' | 'short' | null, facts: string[] }; kind 'answer' without text returns the real unit title as `title` and a one-line prompt as `body`. GET /v1/optionals/outline returns { slug, name, firstPassHours, plannedHours, studied, paused, units: [{ nodeId, title, paper 'I' | 'II' | null, minutes, status 'todo' | 'studied', scheduledOn | null }] } or 404. GET /v1/optionals/status returns { deadline, daysLeft, chosen, reasons, quizTaken } (daysLeft is negative after the deadline).
